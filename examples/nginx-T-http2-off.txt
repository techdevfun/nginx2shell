# configuration file /etc/nginx/nginx.conf:
events {
  worker_connections 1024;
}
http {
  server {
    listen 443 ssl;
    server_name shop.example.com;
    http2 off;
    ssl_certificate /etc/nginx/cert.pem;
    ssl_certificate_key /etc/nginx/key.pem;
    location / {
      proxy_pass http://127.0.0.1:8080;
    }
  }
}
